<Heading level="2">
    模块化引入
</Heading>

<Paragraph>
    为了更小的打包体积，可以按类别或单独引入方法：
</Paragraph>

<Paragraph textAlign="left" blockColor="light_grey">
<Mark bold>javascript</Mark>
<Mark color="grey">// 按类别引入</Mark>
<Mark color="grey">var array = require('lodash/array');</Mark>
<Mark color="grey">var object = require('lodash/fp/object');</Mark>
<Mark color="grey"></Mark>
<Mark color="grey">// 按需引入单个方法（支持 tree-shaking）</Mark>
<Mark color="grey">var at = require('lodash/at');</Mark>
<Mark color="grey">var curryN = require('lodash/fp/curryN');</Mark>
</Paragraph>

<Divider blockColor="light_grey" />

<Heading level="2">
    FP 模块配置
</Heading>

<Paragraph>
    FP 模块通过 <Mark>convert</Mark> 函数支持配置选项：
</Paragraph>

<Table>
    <TableRow>
        <TableCell>
            <Mark bold>选项</Mark>
        </TableCell>
        <TableCell>
            <Mark bold>默认值</Mark>
        </TableCell>
        <TableCell>
            <Mark bold>说明</Mark>
        </TableCell>
    </TableRow>
    <TableRow>
        <TableCell>
            cap
        </TableCell>
        <TableCell>
            true
        </TableCell>
        <TableCell>
            限制 iteratee 的参数数量
        </TableCell>
    </TableRow>
    <TableRow>
        <TableCell>
            curry
        </TableCell>
        <TableCell>
            true
        </TableCell>
        <TableCell>
            启用自动柯里化
        </TableCell>
    </TableRow>
    <TableRow>
        <TableCell>
            fixed
        </TableCell>
        <TableCell>
            true
        </TableCell>
        <TableCell>
            固定参数数量
        </TableCell>
    </TableRow>
    <TableRow>
        <TableCell>
            immutable
        </TableCell>
        <TableCell>
            true
        </TableCell>
        <TableCell>
            不可变操作
        </TableCell>
    </TableRow>
    <TableRow>
        <TableCell>
            rearg
        </TableCell>
        <TableCell>
            true
        </TableCell>
        <TableCell>
            重排参数顺序
        </TableCell>
    </TableRow>
</Table>

<Paragraph textAlign="left" blockColor="light_grey">
<Mark bold>javascript</Mark>
<Mark color="grey">var fp = require('lodash/fp');</Mark>
<Mark color="grey"></Mark>
<Mark color="grey">// 创建自定义配置的 Lodash 实例</Mark>
<Mark color="grey">var custom = fp.runInContext({</Mark>
<Mark color="grey">  'cap': false,</Mark>
<Mark color="grey">  'curry': true</Mark>
<Mark color="grey">});</Mark>
</Paragraph>
